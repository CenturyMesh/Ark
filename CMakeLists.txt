cmake_minimum_required(VERSION 3.20)
project(Ark VERSION 0.7.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(ark STATIC
  src/core.cpp src/nurbs.cpp src/brep.cpp src/mesh.cpp src/csg.cpp
  src/model.cpp src/evaluate.cpp src/solver.cpp src/sketch.cpp src/prism_cut.cpp src/onboarding.cpp src/live_session.cpp src/assistant.cpp src/io.cpp src/serialize.cpp
  src/source_tube.cpp src/notch.cpp src/notch_preview.cpp src/step_export.cpp src/tube.cpp src/assembly.cpp src/compute.cpp src/api.cpp src/mesh_query.cpp src/mesh_analysis.cpp src/agent.cpp src/kernels_scalar.cpp)
target_include_directories(ark PUBLIC include)
target_include_directories(ark SYSTEM PRIVATE third_party)
target_link_libraries(ark PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(ark PRIVATE /W4 /fp:precise)
else()
  target_compile_options(ark PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers -fno-fast-math)
endif()
add_executable(ark-cli src/app/cli.cpp)
add_executable(ark-mcp src/app/mcp_server.cpp)
add_executable(ark-tests tests/tests.cpp)
add_executable(ark-agent-tests tests/agent_tests.cpp)
foreach(app ark-cli ark-mcp ark-tests ark-agent-tests)
  target_link_libraries(${app} PRIVATE ark)
endforeach()
if(WIN32)
  target_link_libraries(ark PUBLIC advapi32 winhttp)
  add_executable(ark-studio WIN32 src/view/viewer.cpp)
  target_link_libraries(ark-studio PRIVATE ark opengl32 gdi32 user32 shell32 comdlg32 dwmapi)
endif()
if(MINGW)
  target_link_options(ark PUBLIC -static)
endif()
enable_testing()
add_test(NAME kernel COMMAND ark-tests)
add_test(NAME agent COMMAND ark-agent-tests)
set_tests_properties(kernel agent PROPERTIES TIMEOUT 60)

find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
  add_test(NAME mcp-stdio COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/mcp_smoke.py $<TARGET_FILE:ark-mcp>)
  set_tests_properties(mcp-stdio PROPERTIES TIMEOUT 30)
endif()

option(ARK_ENABLE_CUDA "Enable optional dynamically loaded CUDA compute on Windows" ON)
if(WIN32 AND ARK_ENABLE_CUDA)
  find_path(ARK_CUDA_INCLUDE cuda.h HINTS "$ENV{CUDA_PATH}/include")
  if(ARK_CUDA_INCLUDE)
    target_include_directories(ark PRIVATE "${ARK_CUDA_INCLUDE}")
    target_compile_definitions(ark PRIVATE ARK_CUDA_RUNTIME=1)
  endif()
endif()
