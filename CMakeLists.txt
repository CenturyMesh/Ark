# Optional at raw configure time; the Windows build scripts prepare the pinned
# local semantic payload before configuring. No model download occurs at runtime.
set(ARK_MEMORY_ASSET_DIRECTORY "${CMAKE_SOURCE_DIR}/out/deps/memory" CACHE PATH "Pinned local semantic model/runtime assets")
if(WIN32 AND EXISTS "${ARK_MEMORY_ASSET_DIRECTORY}/manifest.json")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ARK_MEMORY_ASSET_DIRECTORY}/manifest.json")
  file(READ "${ARK_MEMORY_ASSET_DIRECTORY}/manifest.json" memory_manifest)
  string(JSON memory_file_count LENGTH "${memory_manifest}" files)
  math(EXPR memory_last_file "${memory_file_count}-1")
  set(memory_asset_paths "${ARK_MEMORY_ASSET_DIRECTORY}/manifest.json")
  foreach(memory_index RANGE 0 ${memory_last_file})
    string(JSON memory_file GET "${memory_manifest}" files ${memory_index} path)
    string(JSON memory_expected GET "${memory_manifest}" files ${memory_index} sha256)
    if(NOT memory_file MATCHES "^[A-Za-z0-9_.-]+$" OR memory_file MATCHES "[.][.]")
      message(FATAL_ERROR "Unsafe semantic memory asset path")
    endif()
    file(SHA256 "${ARK_MEMORY_ASSET_DIRECTORY}/${memory_file}" memory_observed)
    if(NOT memory_observed STREQUAL memory_expected)
      message(FATAL_ERROR "Pinned semantic memory asset changed: ${memory_file}")
    endif()
    list(APPEND memory_asset_paths "${ARK_MEMORY_ASSET_DIRECTORY}/${memory_file}")
  endforeach()
  add_custom_target(ark-memory-assets
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:ark-cli>/assets/memory"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${memory_asset_paths} "$<TARGET_FILE_DIR:ark-cli>/assets/memory" VERBATIM)
  foreach(memory_consumer ark-cli ark-mcp ark-studio)
    if(TARGET ${memory_consumer})
      add_dependencies(${memory_consumer} ark-memory-assets)
    endif()
  endforeach()
  add_test(NAME memory-embeddings-semantic COMMAND ark-memory-embeddings-tests "${ARK_MEMORY_ASSET_DIRECTORY}")
  set_tests_properties(memory-embeddings-semantic PROPERTIES TIMEOUT 120)
  add_test(NAME project-memory-semantic COMMAND ark-project-memory-semantic-tests "${ARK_MEMORY_ASSET_DIRECTORY}")
  set_tests_properties(project-memory-semantic PROPERTIES TIMEOUT 120)
endif()
if(WIN32)
  add_test(NAME resource-library-package COMMAND "${CMAKE_COMMAND}" -E env --unset=PSModulePath "${ARK_POWERSHELL}" -NoProfile -NonInteractive -ExecutionPolicy Bypass -File "${CMAKE_SOURCE_DIR}/tests/resource_library_package_tests.ps1" -OutputDirectory "${CMAKE_CURRENT_BINARY_DIR}/resource-library-check")
  add_test(NAME memory-vault COMMAND "${CMAKE_COMMAND}" -E env --unset=PSModulePath "${ARK_POWERSHELL}" -NoProfile -NonInteractive -ExecutionPolicy Bypass -File "${CMAKE_SOURCE_DIR}/tests/memory_vault_tests.ps1")
  set_tests_properties(resource-library-package memory-vault PROPERTIES TIMEOUT 120)
